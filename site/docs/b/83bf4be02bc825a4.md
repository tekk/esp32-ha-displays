```makefile
    ESP_ERROR_CHECK(esp_timer_create(&timer_args, &scan_timer_));	//未修改的108行

```

	is_connecting_ = true;

    vTaskDelay(pdMS_TO_TICKS(500));		//延时，确保接收到WIFI_FAIL_BIT
